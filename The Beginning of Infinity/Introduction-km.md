_វឌ្ឍនភាព_ ដែលលឿនល្មមអាចកត់សម្គាល់បាន និងមានស្ថេរភាពគ្រប់គ្រាន់ក្នុងការបន្តជាច្រើនជំនាន់ត្រូវបានសម្រេចតែម្តងគត់ក្នុងប្រវត្តិសាស្ត្រនៃប្រភេទសត្វរបស់យើង។ វាបានចាប់ផ្តើមនៅប្រហែលពេលនៃបដិវត្តន៍វិទ្យាសាស្ត្រ ហើយកំពុងដំណើរការនៅឡើយ។ វាបានរួមបញ្ចូលការកែលម្អមិនត្រឹមតែនៅក្នុងការយល់ដឹងផ្នែកវិទ្យាសាស្រ្តប៉ុណ្ណោះទេ ប៉ុន្តែក៏មាននៅក្នុងបច្ចេកវិទ្យា ស្ថាប័ននយោបាយ តម្លៃសីលធម៌ សិល្បៈ និងគ្រប់ទិដ្ឋភាពនៃសុខុមាលភាពរបស់មនុស្សផងដែរ។

នៅពេលណាដែលមានការរីកចំរើន មានអ្នកគិតដ៏មានឥទ្ធិពល ដែលបានបដិសេធថាវាជាការពិត ថាវាជាការចង់បាន ឬសូម្បីតែគំនិតនោះមានអត្ថន័យ។ ពួកគេគួរតែដឹងកាន់តែច្បាស់។ ពិតជាមានភាពខុសគ្នានៃគោលបំណងរវាងការពន្យល់មិនពិត និងការពិតមួយ រវាងការបរាជ័យរ៉ាំរ៉ៃក្នុងការដោះស្រាយបញ្ហា និងការដោះស្រាយវា និងរវាងខុស និងត្រូវ អាក្រក់ និងស្រស់ស្អាត ការរងទុក្ខ និងការបន្ធូរបន្ថយរបស់វា ហើយដូច្នេះរវាងការជាប់គាំង និងវឌ្ឍនភាពក្នុងន័យពេញលេញបំផុត។

នៅក្នុងសៀវភៅនេះ ខ្ញុំបានប្រកែកថា វឌ្ឍនភាពទាំងអស់ ទាំងទ្រឹស្តី និងការអនុវត្ត បានមកពីសកម្មភាពរបស់មនុស្សតែមួយ៖ ការស្វែងរកអ្វីដែលខ្ញុំហៅថា ការពន្យល់ដ៏ល្អ។ ទោះបីជាដំណើរស្វែងរកនេះមានលក្ខណៈប្លែកពីមនុស្សក៏ដោយ ប្រសិទ្ធភាពរបស់វាក៏ជាការពិតជាមូលដ្ឋានអំពីការពិតនៅកម្រិតលោហធាតុដែលមិនមានលក្ខណៈបុគ្គលបំផុតដែរ ពោលគឺវាអនុលោមតាមច្បាប់សកលនៃធម្មជាតិដែលពិតជាការពន្យល់ដ៏ល្អ។ ទំនាក់ទំនងដ៏សាមញ្ញនេះ រវាងលោហធាតុ និងមនុស្ស គឺជាតម្រុយនៃតួនាទីកណ្តាលរបស់ _មនុស្ស_ នៅក្នុងគ្រោងការណ៍លោហធាតុនៃវត្ថុ។

ត្រូវតែឈានទៅដល់ទីបញ្ចប់ – ទាំងនៅក្នុងមហន្តរាយ ឬនៅក្នុងការបញ្ចប់មួយចំនួន – ឬវាគ្មានដែនកំណត់? ចម្លើយគឺចុងក្រោយ។ ភាពគ្មានព្រំដែនគឺជា 'ភាពគ្មានទីបញ្ចប់' ដែលសំដៅលើចំណងជើងនៃសៀវភៅនេះ។ ការពន្យល់វា និងលក្ខខណ្ឌដែលវឌ្ឍនភាពអាច និងមិនអាចកើតឡើងបាន រួមបញ្ចូលដំណើរឆ្លងកាត់ស្ទើរតែគ្រប់វិស័យជាមូលដ្ឋាននៃវិទ្យាសាស្ត្រ និងទស្សនវិជ្ជា។ ពីវិស័យនីមួយៗ យើងរៀនថា ទោះបីជាការរីកចំរើនមិនមានទីបញ្ចប់ក៏ដោយ វាមានការចាប់ផ្តើមចាំបាច់មួយ៖ បុព្វហេតុ ឬព្រឹត្តិការណ៍ដែលវាចាប់ផ្តើម ឬលក្ខខណ្ឌចាំបាច់សម្រាប់វាដើម្បីឈានចេញ និងរីកចម្រើន។ រាល់ការចាប់ផ្តើមទាំងនេះគឺជា 'ការចាប់ផ្តើមនៃភាពគ្មានទីបញ្ចប់' ដូចដែលបានមើលពីទស្សនៈនៃវាលនោះ។ មនុស្សជាច្រើនហាក់ដូចជាមិនទាក់ទងគ្នាទេ។ ប៉ុន្តែពួកគេទាំងអស់គឺជាមុខនៃគុណលក្ខណៈតែមួយនៃការពិត ដែលខ្ញុំហៅថា _the_ ការចាប់ផ្តើមនៃភាពគ្មានទីបញ្ចប់។
